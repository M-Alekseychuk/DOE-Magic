+// Normal inverse CDF approximation
+function normSInv(p: number): number {
+  if (p >= 1) return 3.5;
+  if (p <= 0) return -3.5;
+  const a1 = -39.6968302866538;
+  const a2 = 220.946098424521;
+  const a3 = -275.928510446969;
+  const a4 = 138.357751867269;
+  const a5 = -30.6647980661472;
+  const a6 = 2.50662827745924;
+  const b1 = -54.4760987982241;
+  const b2 = 161.585836858041;
+  const b3 = -155.698979859887;
+  const b4 = 66.8013118877197;
+  const b5 = -13.2806815528857;
+  const c1 = -0.00778489400243029;
+  const c2 = -0.322396458041136;
+  const c3 = -2.40075827716184;
+  const c4 = -2.54973253934373;
+  const c5 = 4.37466414146497;
+  const c6 = 2.93816398269878;
+  const d1 = 0.00778469570904146;
+  const d2 = 0.32246712907004;
+  const d3 = 2.445134137143;
+  const d4 = 3.75440866190742;
+
+  let q, r, x;
+
+  if (p < 0.02425) {
+      q = Math.sqrt(-2 * Math.log(p));
+      x = (((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6) /
+          ((((d1 * q + d2) * q + d3) * q + d4) * q + 1);
+  } else if (p <= 0.97575) {
+      q = p - 0.5;
+      r = q * q;
+      x = (((((a1 * r + a2) * r + a3) * r + a4) * r + a5) * r + a6) * q /
+          (((((b1 * r + b2) * r + b3) * r + b4) * r + b5) * r + 1);
+  } else {
+      q = Math.sqrt(-2 * Math.log(1 - p));
+      x = -(((((c1 * q + c2) * q + c3) * q + c4) * q + c5) * q + c6) /
+          ((((d1 * q + d2) * q + d3) * q + d4) * q + 1);
+  }
+
+  return x;
+}

+// Matrix math utilities for OLS
+function transpose(m: number[][]) {
+    return m[0].map((_, i) => m.map(row => row[i]));
+}

+function multiply(m1: number[][], m2: number[][]) {
+    const res = Array(m1.length).fill(0).map(() => Array(m2[0].length).fill(0));
+    for (let i = 0; i < m1.length; i++) {
+        for (let j = 0; j < m2[0].length; j++) {
+            let sum = 0;
+            for (let k = 0; k < m1[0].length; k++) {
+                sum += m1[i][k] * m2[k][j];
+            }
+            res[i][j] = sum;
+        }
+    }
+    return res;
+}

+function solveOLS(X: number[][], Y: number[][]) {
+    const n = X.length;
+    const k = X[0].length;
+    
+    const Xt = transpose(X);
+    const XtX = multiply(Xt, X);
+    const XtY = multiply(Xt, Y);
+    
+    // Augment: [ XtX | XtY | I ]
+    const aug: number[][] = [];
+    for (let i = 0; i < k; i++) {
+        const row = [...XtX[i], XtY[i][0]];
+        for (let j = 0; j < k; j++) row.push(i === j ? 1 : 0);
+        aug.push(row);
+    }
+    
+    const aliased = new Array(k).fill(false);
+    let currentRow = 0;
+    const colToRow = new Array(k).fill(-1);
+    
+    for (let col = 0; col < k; col++) {
+        let pivotRow = currentRow;
+        let maxVal = 0;
+        for (let r = currentRow; r < k; r++) {
+            if (Math.abs(aug[r][col]) > maxVal) {
+                maxVal = Math.abs(aug[r][col]);
+                pivotRow = r;
+            }
+        }
+        
+        if (maxVal < 1e-9) {
+            aliased[col] = true;
+            continue;
+        }
+        
+        if (pivotRow !== currentRow) {
+            const temp = aug[currentRow];
+            aug[currentRow] = aug[pivotRow];
+            aug[pivotRow] = temp;
+        }
+        
+        const pivot = aug[currentRow][col];
+        for (let c = col; c < aug[currentRow].length; c++) {
+            aug[currentRow][c] /= pivot;
+        }
+        
+        for (let r = 0; r < k; r++) {
+            if (r !== currentRow) {
+                const factor = aug[r][col];
+                for (let c = col; c < aug[r].length; c++) {
+                    aug[r][c] -= factor * aug[currentRow][c];
+                }
+            }
+        }
+        
+        colToRow[col] = currentRow;
+        currentRow++;
+    }
+    
+    const rank = currentRow;
+    const beta = new Array(k).fill(0);
+    const diagInv = new Array(k).fill(0);
+    const invXtX: number[][] = Array(k).fill(0).map(() => Array(k).fill(0));
+    
+    for (let col = 0; col < k; col++) {
+        if (!aliased[col]) {
+            const r = colToRow[col];
+            beta[col] = aug[r][k];
+            diagInv[col] = Math.max(0, aug[r][k + 1 + col]);
+            for (let c2 = 0; c2 < k; c2++) {
+                if (!aliased[c2]) {
+                    invXtX[col][c2] = aug[r][k + 1 + c2];
+                }
+            }
+        }
+    }
+    
+    return { beta, diagInv, invXtX, aliased, rank };
+}

+        { x: 0, y: 0 },
+        { x: maxT, y: halfNormalSlope * maxT }
+    ];
+
+    const paretoData = [...combinedEffectsForPlot].sort((a, b) => b.tRatio - a.tRatio);
+    const bonferroniLimitPareto = df_error > 0 && activeTerms.length > 0 ? Math.abs(jStat.studentt.inv(1 - 0.05 / (2 * activeTerms.length), df_error)) : 3.5;
+
+
+    // 8. Predictive Model Optimization & Statistical Inference (JMP / Design-Expert Standard)
+    // Predictor helper for any coded factor vector in [-1, 1]^k
+    const buildModelRow = (coded: number[]) => {
+      const row = [1];
+      activeTerms.forEach(t => {
+        if (t.type === 'main') {
+          row.push(coded[t.idx]);
+        } else if (t.type === 'interaction') {
+          row.push(coded[t.i] * coded[t.j]);
+        }
+      });
+      return row;
+    };
+
+    const predictAt = (coded: number[]) => {
+      const row = buildModelRow(coded);
+      let y_hat = 0;
+      for (let idx = 0; idx < row.length; idx++) {
+        if (!aliased[idx]) {
+          y_hat += beta[idx] * row[idx];
+        }
+      }
+      return y_hat;
+    };
+
+    // Calculate prediction leverage, standard errors, and confidence intervals
+    const getPointStats = (coded: number[]) => {
+      const row = buildModelRow(coded);
+      let h00 = 0;
+      const numCols = row.length;
+      for (let r1 = 0; r1 < numCols; r1++) {
+        if (aliased[r1]) continue;
+        for (let r2 = 0; r2 < numCols; r2++) {
+          if (aliased[r2]) continue;
+          h00 += row[r1] * invXtX[r1][r2] * row[r2];
+        }
+      }
+      h00 = Math.max(0, h00);
+      const seMean = Math.sqrt(Math.max(0, mse * h00));
+      const sePred = Math.sqrt(Math.max(0, mse * (1 + h00)));
+      const tCrit = df_error > 0 ? (jStat.studentt.inv(0.975, df_error) || 1.96) : 1.96;
+      const y_hat = predictAt(coded);
+      return {
+        y_hat,
+        h00,
+        seMean,
+        sePred,
+        tCrit,
+        ciLower: y_hat - tCrit * seMean,
+        ciUpper: y_hat + tCrit * seMean,
+        piLower: y_hat - tCrit * sePred,
+        piUpper: y_hat + tCrit * sePred
+      };
+    };
+
+    // Standard Derringer-Suich desirability function
+    const observedY = validRuns.map(r => Number(r.responses[activeResponse.id]));
+    const yMin = Math.min(...observedY);
+    const yMax = Math.max(...observedY);
+    const ySpan = yMax - yMin > 1e-9 ? yMax - yMin : 1;
+    const targetVal = effectiveTargetValue !== undefined && effectiveTargetValue !== null ? effectiveTargetValue : overallMean;
+
+    const calcDesirabilityAndScore = (y_hat: number): { d: number, score: number } => {
+      if (effectiveGoal === 'maximize') {
+        const upperTarget = effectiveTargetValue !== undefined && effectiveTargetValue !== null ? effectiveTargetValue : yMax;
+        const d = upperTarget > yMin ? Math.max(0, Math.min(1, (y_hat - yMin) / (upperTarget - yMin))) : 1;
+        const score = d * 1000 + y_hat;
+        return { d, score };
+      } else if (effectiveGoal === 'minimize') {
+        const lowerTarget = effectiveTargetValue !== undefined && effectiveTargetValue !== null ? effectiveTargetValue : yMin;
+        const d = yMax > lowerTarget ? Math.max(0, Math.min(1, (yMax - y_hat) / (yMax - lowerTarget))) : 1;
+        const score = d * 1000 - y_hat;
+        return { d, score };
+      } else {
+        const tol = ySpan * 0.5;
+        const dist = Math.abs(y_hat - targetVal);
+        const d = Math.max(0, Math.min(1, 1 - dist / tol));
+        const score = d * 1000 - dist;
+        return { d, score };
+      }
+    };
+
+    // Evaluate candidate points:
+    // 1. All 2^k extreme corners
+    let bestCoded: number[] = new Array(k).fill(0);
+    let bestScore = -Infinity;
+    let bestD = 0;
+    let bestPred = overallMean;
+
+    const numCorners = Math.min(1024, Math.pow(2, k));
+    for (let c = 0; c < numCorners; c++) {
+      const coded: number[] = [];
+      for (let i = 0; i < k; i++) {
+        coded.push(((c >> i) & 1) === 1 ? 1 : -1);
+      }
+      const y_hat = predictAt(coded);
+      const { d, score } = calcDesirabilityAndScore(y_hat);
+      if (score > bestScore) {
+        bestScore = score;
+        bestD = d;
+        bestPred = y_hat;
+        bestCoded = coded;
+      }
+    }
+
+    // 2. Center point (0, 0, ... 0)
+    {
+      const centerCoded = new Array(k).fill(0);
+      const yCenter = predictAt(centerCoded);
+      const { d, score } = calcDesirabilityAndScore(yCenter);
+      if (score > bestScore) {
+        bestScore = score;
+        bestD = d;
+        bestPred = yCenter;
+        bestCoded = centerCoded;
+      }
+    }
+
+    // 3. Local coordinate refinement to locate continuous optima (crucial for Target responses)
+    let refinedCoded = [...bestCoded];
+    let improved = true;
+    let iter = 0;
+    while (improved && iter < 12) {
+      improved = false;
+      iter++;
+      for (let i = 0; i < k; i++) {
+        const step = 0.05;
+        const candidates = [
+          Math.min(1, refinedCoded[i] + step),
+          Math.max(-1, refinedCoded[i] - step)
+        ];
+        for (const cand of candidates) {
+          const testCoded = [...refinedCoded];
+          testCoded[i] = cand;
+          const y_test = predictAt(testCoded);
+          const { d, score } = calcDesirabilityAndScore(y_test);
+          if (score > bestScore + 1e-6) {
+            bestScore = score;
+            bestD = d;
+            bestPred = y_test;
+            refinedCoded = testCoded;
+            improved = true;
+          }
+        }
+      }
+    }
+    bestCoded = refinedCoded;
+
+    // Calculate statistical inference metrics at optimum
+    const optStats = getPointStats(bestCoded);
+
+    // Compute factor optimal settings and Marginal Profiler Slopes (dY/dX)
+    const optimalFactors = project.factors.map((f, i) => {
+      const codedVal = bestCoded[i];
+      const naturalVal = f.min + ((codedVal + 1) / 2) * (f.max - f.min);
+
+      // Marginal Profiler Slope: dY/dX at optimum
+      let codedSlope = 0;
+      activeTerms.forEach((t, tIdx) => {
+        const betaIdx = 1 + tIdx;
+        if (aliased[betaIdx]) return;
+        if (t.type === 'main' && t.idx === i) {
+          codedSlope += beta[betaIdx];
+        } else if (t.type === 'interaction') {
+          if (t.i === i) {
+            codedSlope += beta[betaIdx] * bestCoded[t.j];
+          } else if (t.j === i) {
+            codedSlope += beta[betaIdx] * bestCoded[t.i];
+          }
+        }
+      });
+
+      const halfRange = (f.max - f.min) / 2;
+      const naturalSlope = halfRange > 0 ? codedSlope / halfRange : codedSlope;
+
+      let codedLabel = `${codedVal >= 0 ? '+' : ''}${codedVal.toFixed(2)}`;
+      if (codedVal >= 0.95) codedLabel = 'High (+1.00)';
+      else if (codedVal <= -0.95) codedLabel = 'Low (-1.00)';
+      else if (Math.abs(codedVal) <= 0.05) codedLabel = 'Center (0.00)';
+
+      return {
+        factor: f,
+        optimalValue: Number(naturalVal.toFixed(naturalVal % 1 === 0 ? 1 : 2)),
+        codedValue: codedVal,
+        codedLabel,
+        codedSlope,
+        naturalSlope,
+        effectValue: codedSlope * 2
+      };
+    });
+
+    const optimalConditions = {
+      expectedY: optStats.y_hat,
+      desirability: Math.max(0, Math.min(1, bestD)),
+      seFit: optStats.seMean,
+      sePred: optStats.sePred,
+      leverage: optStats.h00,
+      tCrit: optStats.tCrit,
+      ciLower: optStats.ciLower,
+      ciUpper: optStats.ciUpper,
+      piLower: optStats.piLower,
+      piUpper: optStats.piUpper,
+      rmse: Math.sqrt(mse),
+      rSquared,
+      adjRSquared,
+      fRatio: f_model,
+      pModel: p_model,
+      dfError: df_error,
+      factors: optimalFactors
+    };
+
+    const optimalSettings = optimalFactors;
+
+    const parameterEstimates = [];
+    for (let i = 0; i < numTerms; i++) {
+        const stdError = Math.sqrt(mse * diagInv[i]);
+        const tRatio = stdError > 0 ? beta[i] / stdError : 0;
+        let pVal = 1;
+        if (df_error > 0 && stdError > 0) {
+            pVal = (1 - jStat.studentt.cdf(Math.abs(tRatio), df_error)) * 2;
+        }
+        
+        let termName = "Intercept";
+        if (i > 0) {
+            const term = activeTerms[i - 1];
+            if (term && term.type === 'main') {
+                termName = `${term.factor.name}(${term.factor.min},${term.factor.max})`;
+            } else if (term && term.type === 'interaction') {
+                termName = `${term.f1.name}*${term.f2.name}`;
+            }
+        }
+        
+        parameterEstimates.push({
+            term: termName,
+            estimate: beta[i],
+            stdError,
+            tRatio,
+            pVal,
+            isAliased: aliased[i]
+        });
+    }
+
+    // Calculate Actual vs Predicted and Studentized Residuals
+    const actualByPredicted: any[] = [];
+    const studentizedResiduals: any[] = [];
+    const h: number[] = [];
+    
+    // tCrit for 95% limit in studentized residuals (approximate with t-dist or normal if large df)
+    // using jStat for accurate studentized limit (Bonferroni corrected or not)
+    const tLimit = df_error > 0 ? Math.abs(jStat.studentt.inv(1 - 0.05 / 2, df_error)) : 3;
+    const bonferroniLimit = df_error > 0 ? Math.abs(jStat.studentt.inv(1 - 0.05 / (2 * nRuns), df_error)) : 3;
+
+    validRuns.forEach((run, idx) => {
+        const actual = Y[idx][0];
+        let pred = 0;
+        for (let i = 0; i < numTerms; i++) {
+            pred += beta[i] * X[idx][i];
+        }
+        
+        // Calculate leverage h_ii
+        let hii = 0;
+        for (let i = 0; i < numTerms; i++) {
+            if (aliased[i]) continue;
+            let sum = 0;
+            for (let j = 0; j < numTerms; j++) {
+                if (aliased[j]) continue;
+                sum += X[idx][i] * invXtX[i][j] * X[idx][j];
+            }
+            hii += sum;
+        }
+        h.push(hii);
+        
+        const residual = actual - pred;
+        let studentized = 0;
+        if (mse > 1e-9 && hii < 0.9999) {
+            studentized = residual / Math.sqrt(mse * (1 - hii));
+        }
+        
+        actualByPredicted.push({ actual, predicted: pred });
+        studentizedResiduals.push({ run: idx + 1, studentized, limit: tLimit, bonferroniLimit });
+    });
+
+    return { 
+      response: activeResponse, 
+      effects, 
+      paretoData, 
+      optimalSettings, 
+      optimalConditions,
+      halfNormalData, 
+      halfNormalLine,
+      bestPred, 
+      rSquared, 
+      adjRSquared, 
+      p: p_model, 
+      nRuns, 
+      paretoThreshold, 
+      tLimitPareto,
+      bonferroniLimitPareto,
+      effectSummary, 
+      lackOfFit, 
+      parameterEstimates,
+      rmse: Math.sqrt(mse),
+      fRatio: f_model,
+      dfError: df_error,
+      mse,
+      actualByPredicted,
+      studentizedResiduals
+    };
+  }, [project, hasData, activeResponse, excludedTerms, effectiveGoal, effectiveTargetValue]);
+
+  const interactionData = useMemo(() => {
+    if (!hasData || project.factors.length < 2 || !activeResponse) return null;
+    const validRuns = project.runs.filter(r => r.responses[activeResponse.id] !== null);
+    
+    let f1 = project.factors.find(f => f.id === interactionF1Id);
+    let f2 = project.factors.find(f => f.id === interactionF2Id);
+    
+    if (!f1) f1 = project.factors[0];
+    if (!f2 || f2.id === f1.id) {
+       f2 = project.factors.find(f => f.id !== f1?.id) || project.factors[1];
+    }
+    if (!f1 || !f2) return null;
+
+    const f1Mid = (f1.min + f1.max) / 2;
+    const f2Mid = (f2.min + f2.max) / 2;
+
+    let llSum = 0, llCount = 0;
+    let lhSum = 0, lhCount = 0;
+    let hlSum = 0, hlCount = 0;
+    let hhSum = 0, hhCount = 0;
+
+    validRuns.forEach(run => {
+      const v1 = Number(run.factors[f1!.id]);
+      const v2 = Number(run.factors[f2!.id]);
+      const res = Number(run.responses[activeResponse.id]);
+
+      if (v1 < f1Mid - 1e-6 && v2 < f2Mid - 1e-6) { llSum += res; llCount++; }
+      if (v1 < f1Mid - 1e-6 && v2 > f2Mid + 1e-6) { lhSum += res; lhCount++; }
+      if (v1 > f1Mid + 1e-6 && v2 < f2Mid - 1e-6) { hlSum += res; hlCount++; }
+      if (v1 > f1Mid + 1e-6 && v2 > f2Mid + 1e-6) { hhSum += res; hhCount++; }
+    });
+
+    return {
+      f1, f2,
+      data: [
+        { x: f1.min, low: llCount > 0 ? llSum / llCount : null, high: lhCount > 0 ? lhSum / lhCount : null },
+        { x: f1.max, low: hlCount > 0 ? hlSum / hlCount : null, high: hhCount > 0 ? hhSum / hhCount : null },
+      ]
+    };
+  }, [project, hasData, activeResponse, interactionF1Id, interactionF2Id]);
+
+  // DOE Mean plot data
+  const doeMeanData = useMemo(() => {
+    if (!hasData || !activeResponse) return null;
+    return project.runs
+      .filter(r => r.responses[activeResponse.id] !== null)
+      .map((r, i) => ({
+        run: `Run ${i + 1}`,
+        value: r.responses[activeResponse.id]
+      }));
+  }, [project, hasData, activeResponse]);
+
+  // 3D Plot data
+  const plotData = useMemo(() => {
+    if (!hasData || project.factors.length < 2 || !activeResponse) return null;
+    
+    const xFactor = project.factors.find(f => f.id === xAxisFactorId) || project.factors[0];
+    const yFactor = project.factors.find(f => f.id === yAxisFactorId && f.id !== xFactor.id) || project.factors[1] || project.factors[0];
+    
+    const validRuns = project.runs.filter(r => r.responses[activeResponse.id] !== null);
+    
+    if (plot3dType === 'scatter3d') {
+      const xValues = validRuns.map(r => r.factors[xFactor.id]);
+      const yValues = validRuns.map(r => r.factors[yFactor.id]);
+      const zValues = validRuns.map(r => r.responses[activeResponse.id]);
+      return {
+        xFactor, yFactor, activeResponse,
+        data: [{
+          x: xValues,
+          y: yValues,
+          z: zValues,
+          type: 'scatter3d',
+          mode: 'markers',
+          marker: { size: 8, color: zValues, colorscale: 'Viridis', opacity: 0.8 }
+        }]
+      };
+    }
+
+    // For surface and contour, build a surrogate model and evaluate on a grid
+    const k = project.factors.length;
+    const X = validRuns.map(r => project.factors.map(f => Number(r.factors[f.id])));
+    const Y = validRuns.map(r => Number(r.responses[activeResponse.id]));
+    
+    const meanX = Array(k).fill(0);
+    let meanY = 0;
+    for (let i = 0; i < X.length; i++) {
+      meanY += Y[i];
+      for (let j = 0; j < k; j++) meanX[j] += X[i][j];
+    }
+    meanY /= X.length;
+    for (let j = 0; j < k; j++) meanX[j] /= X.length;
+
+    const beta = Array(k).fill(0);
+    for (let j = 0; j < k; j++) {
+      let cov = 0;
+      let varX = 0;
+      for (let i = 0; i < X.length; i++) {
+        cov += (X[i][j] - meanX[j]) * (Y[i] - meanY);
+        varX += Math.pow(X[i][j] - meanX[j], 2);
+      }
+      beta[j] = varX > 0 ? cov / varX : 0;
+    }
+
+    const getTrend = (x: number[]) => {
+      let y = meanY;
+      for (let j = 0; j < k; j++) y += beta[j] * (x[j] - meanX[j]);
+      return y;
+    };
+    
+    const residuals = X.map((xi, i) => Y[i] - getTrend(xi));
+
+    // Normalize distances based on factor ranges to prevent skewing
+    const ranges = project.factors.map(f => Math.max(1e-5, f.max - f.min));
+
+    const predict = (x: number[]) => {
+      let num = 0;
+      let den = 0;
+      for (let i = 0; i < X.length; i++) {
+        let dSq = 0;
+        for (let j = 0; j < k; j++) {
+          dSq += Math.pow((x[j] - X[i][j]) / ranges[j], 2);
+        }
+        if (dSq < 1e-10) return Y[i];
+        const w = 1 / (dSq + 1e-8);
+        num += w * residuals[i];
+        den += w;
+      }
+      return getTrend(x) + (den > 0 ? (num / den) : 0);
+    };
+    
+    const gridSize = 35;
+    const xMin = xFactor.min;
+    const xMax = xFactor.max;
+    const yMin = yFactor.min;
+    const yMax = yFactor.max;
+    
+    const gridX = Array.from({length: gridSize}, (_, i) => xMin + (xMax - xMin) * i / (gridSize - 1));
+    const gridY = Array.from({length: gridSize}, (_, i) => yMin + (yMax - yMin) * i / (gridSize - 1));
+    const zGrid: number[][] = [];
+    
+    for (let i = 0; i < gridSize; i++) {
+      const row: number[] = [];
+      const currentY = gridY[i];
+      for (let j = 0; j < gridSize; j++) {
+        const currentX = gridX[j];
+        
+        const x_vec = project.factors.map(f => {
+          if (f.id === xFactor.id) return currentX;
+          if (f.id === yFactor.id) return currentY;
+          return (f.min + f.max) / 2;
+        });
+        
+        row.push(predict(x_vec));
+      }
+      zGrid.push(row);
+    }
+
+    if (plot3dType === 'mesh3d') {
+      return {
+        xFactor, yFactor, activeResponse,
+        data: [{
+          x: gridX,
+          y: gridY,
+          z: zGrid,
+          type: 'surface',
+          colorscale: 'Viridis',
+          contours: {
+            z: {
+              show: true,
+              usecolormap: false,
+              color: "#334155",
+              highlightcolor: "#0f172a",
+              project: { z: true }
+            }
+          }
+        }]
+      };
+    } else if (plot3dType === 'contour') {
+      return {
+        xFactor, yFactor, activeResponse,
+        data: [{
+          x: gridX,
+          y: gridY,
+          z: zGrid,
+          type: 'contour',
+          colorscale: 'Viridis',
+        }]
+      };
+    }
+    
+    return null;
+  }, [project, hasData, activeResponse, xAxisFactorId, yAxisFactorId, plot3dType]);
+
+  if (!hasData) {
+    return <div className="p-8 text-center text-[11px] text-slate-500 dark:text-slate-300 bg-white dark:bg-slate-900 rounded-lg shadow-sm border border-slate-200 dark:border-slate-600">Insufficient data for analysis. Please add factors, responses, and run results.</div>;
+  }
+
+  return (
+    <div className="space-y-4">
+      {/* Analysis Controls */}
+      <div className="bg-white dark:bg-slate-900 p-3 rounded-lg shadow-sm border border-slate-200 dark:border-slate-600 flex flex-wrap items-center gap-6">
+        {project.responses.length > 1 && (
+          <div className="flex items-center gap-3">
+            <span className="text-[10px] font-bold text-slate-500 dark:text-slate-300 uppercase tracking-widest">Analyze Response:</span>
+            <select 
+              value={activeResponseId} 
+              onChange={e => onResponseChange(e.target.value)}
+              className="rounded border border-slate-300 dark:border-slate-600 px-2 py-1 text-[11px] focus:outline-none focus:ring-1 focus:ring-brand-500"
+            >
+              {project.responses.map(r => (
+                <option key={r.id} value={r.id}>{r.name} ({r.goal})</option>
+              ))}
+            </select>
+          </div>
+        )}
+        <div className="flex items-center gap-2 text-[11px] text-slate-500 dark:text-slate-400">
+          <span className="text-[10px] font-bold text-slate-600 dark:text-slate-300 uppercase tracking-wider">Effects Evaluated:</span>
+          <span className="inline-flex items-center gap-1.5 px-2 py-0.5 rounded bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300 font-medium text-[11px] border border-slate-200 dark:border-slate-700">
+            All (Primary + Secondary)
+          </span>
+        </div>
+      </div>
+
+      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
+        
+        {/* Effects Plot (Merged Pareto & Half-Normal) */}
+        <div id="effects-plot" className="bg-white dark:bg-slate-900 p-4 rounded-lg shadow-sm border border-slate-200 dark:border-slate-600 flex flex-col">
+          <div className="flex justify-between items-start mb-4">
+            <div>
+              <div className="flex items-center gap-3 mb-1">
+                <h3 className="text-[10px] uppercase tracking-widest font-bold text-slate-400">
+                  Effects Plot <span className="normal-case font-normal">({activeResponse?.name})</span>
+                </h3>
+                <select 
+                  value={effectsPlotType}
+                  onChange={e => setEffectsPlotType(e.target.value as 'pareto' | 'half-normal')}
+                  className="rounded border border-slate-300 dark:border-slate-600 px-1 py-0.5 text-[10px] focus:outline-none focus:ring-1 focus:ring-brand-500 bg-slate-50 dark:bg-slate-800"
+                >
+                  <option value="pareto">Pareto: Standardized</option>
+                  <option value="half-normal">Half-Normal</option>
+                </select>
+              </div>
+              <p className="text-[10px] text-slate-500">
+                {effectsPlotType === 'pareto' 
+                  ? 'Shows the absolute size of each effect. Bars extending further indicate a stronger impact on the response.' 
+                  : 'Helps identify significant effects. Points that deviate furthest from the line represent factors with a meaningful impact.'}
+              </p>
+            </div>
+            <button
+              onClick={() => handleDownloadSvg('effects-plot', `${project.name}_${effectsPlotType}_Plot`)}
+              className="p-1.5 text-slate-400 hover:text-brand-600 hover:bg-slate-100 dark:hover:bg-slate-800 rounded transition-colors"
+              title="Download PNG"
+            >
+              <Download size={14} />
+            </button>
+          </div>
